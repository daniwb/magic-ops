		source.Owner, source.Controller = 0, 0

		obj := &StackObject{
			IsAbility:  true,

			Controller: 0,
			Card:       source,
			EffectValue: map[string]interface{}{
				"power":      3,
				"toughness": 0,
				"target_ref": "self",
			},
		}
		if err := gs.ExecuteAbilityEffect(obj); err != nil {
			t.Fatalf("ExecuteAbilityEffect: %v", err)
		}
		if got := source.GetPowerWithEffects(gs.EffectManager); got != 3 {
			t.Fatalf("source power = %d, want 3", got)
		}
	})
