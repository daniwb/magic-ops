func TestFactoryNGDynamicAmountManaValueOfExiledThisWay(t *testing.T) {

		card.Type = "Artifact"
		card.Types = []string{"Artifact"}
		return card
	}

	source := NewCreature("Test Spell", ManaCost{}, 1, 1, 0)
	count := map[string]interface{}{"count": "exiled_this_way_mana_value"}

	t.Run("successful exile feeds later consumer and is consumed once", func(t *testing.T) {
		gs := NewGame("P0", "P1", false)
		artifact := newArtifact("Test Relic", 3)
		gs.Battlefield.AddCard(artifact)
		life := gs.Players[0].Life

		err := gs.ExecuteEffectSequence(0, source, []EffectAtomSpec{
			{Effect: "exile"},
			{Effect: "gain_life", EffectValue: map[string]interface{}{"amount_count": count}},
			{Effect: "gain_life", EffectValue: map[string]interface{}{"amount_count": count}},
		}, []Target{{IsCard: true, CardID: artifact.ID}}, 0, nil)
		if err != nil {
			t.Fatal(err)
		}
		if got := gs.Players[0].Life; got != life+3 {
			t.Fatalf("life = %d, want %d", got, life+3)
		}
	})

		gs := NewGame("P0", "P1", false)
