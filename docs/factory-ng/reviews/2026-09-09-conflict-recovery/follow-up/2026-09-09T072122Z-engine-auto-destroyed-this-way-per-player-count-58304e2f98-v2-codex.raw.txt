func (gs *GameState) executeDestroyEffect(targets []Target, cantBeRegenerated bool) error {
	gs.resetDestroyedArtifactTally()
	destroy := DestroyTarget
	if cantBeRegenerated {
		destroy = DestroyTargetForced
	}
	for _, target := range targets {
		gs.destroyAndNoteArtifact(target, destroy)
	}
	return nil
}

func (gs *GameState) resetDestroyedArtifactTally() {
	if gs.seqTally == nil {
		return
	}
	for key := range gs.seqTally {
		if strings.HasPrefix(key, "destroyed_artifact_p") {
			delete(gs.seqTally, key)
		}
	}
}

func (gs *GameState) destroyAndNoteArtifact(target Target, destroy func(Target, *GameState) error) {
	if gs.seqTally == nil || !target.IsCard || target.CardID == "" {
		_ = destroy(target, gs)
		return
	}
	card, err := gs.Battlefield.GetCard(target.CardID)
	if err != nil || card == nil {
		return
	}
	controller := card.Controller
	isArtifact := card.HasType("Artifact") || card.HasSubType("Artifact")
	if err := destroy(target, gs); err != nil || !isArtifact {
		return
	}
	if card.Owner >= 0 && card.Owner < len(gs.Players) &&
		gs.Players[card.Owner].Graveyard.HasCard(card.ID) {
		key := "destroyed_artifact_p" + strconv.Itoa(controller)
		gs.seqTally[key]++
	}
}
