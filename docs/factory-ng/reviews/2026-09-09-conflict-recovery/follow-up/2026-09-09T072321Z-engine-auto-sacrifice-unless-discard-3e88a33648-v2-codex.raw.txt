	SourceID  string
}

func (gs *GameState) executeSacrificeUnlessDiscard(controller int, source *Card) error {
	if controller < 0 || controller >= len(gs.Players) || source == nil {
		return nil
	}
	if _, err := gs.Battlefield.GetCard(source.ID); err != nil {
		return nil
	}
	gs.PendingSacrificeUnlessDiscard = &PendingSacrificeUnlessDiscard{
		PlayerIdx: controller,
		SourceID:  source.ID,
	}
	return nil
}

func (gs *GameState) ResolveSacrificeUnlessDiscard(playerIdx int, discardCardID string) error {
	pending := gs.PendingSacrificeUnlessDiscard
	if pending == nil || pending.PlayerIdx != playerIdx {
		return nil
	}
	if playerIdx < 0 || playerIdx >= len(gs.Players) {
		return fmt.Errorf("invalid player")
	}
	if discardCardID != "" && !gs.Players[playerIdx].Hand.HasCard(discardCardID) {
		return fmt.Errorf("card %s not found in hand", discardCardID)
	}

	gs.PendingSacrificeUnlessDiscard = nil
	if discardCardID != "" {
		return gs.Discard(playerIdx, discardCardID)
	}

	if source, err := gs.Battlefield.GetCard(pending.SourceID); err == nil {
		gs.sacrificeCard(source, playerIdx)
		gs.CheckStateBasedActions()
	}
	return nil
}

package cards

import (
	"testing"

	"magic-backend/game"
)

func TestFactoryNGSacrificeUnlessDiscard(t *testing.T) {
	newCard := func(t *testing.T) *game.Card {
		t.Helper()
		power, toughness := 4, 4
		def := &CardDefinition{
			Name: "Factory NG Masticore",
			Type: "Creature",
			Types: []string{"Creature"},
			Power: &power,
			Toughness: &toughness,
			Abilities: []AbilityDSL{{
				Type:    "triggered",
				Trigger: "upkeep",
				Effect:  "sacrifice_unless_discard",
			}},
		}
		card, err := def.ToGameCard(0)
		if err != nil {
			t.Fatalf("ToGameCard: %v", err)
		}
		return card
	}

	t.Run("public trigger and resolution path", func(t *testing.T) {
		gs := game.NewGame("P0", "P1", false)
		src := newCard(t)
		src.Owner, src.Controller = 0, 0
		gs.Battlefield.AddCard(src)
		gs.RegisterCardAbilities(src)
		gs.ProcessTriggeredAbilities(game.CreateUpkeepEvent(0))
		if err := gs.ResolveTopOfStack(); err != nil {
			t.Fatalf("ResolveTopOfStack: %v", err)
		}
		if gs.PendingSacrificeUnlessDiscard == nil {
			t.Fatal("expected pending sacrifice-unless-discard choice")
		}

		err := gs.ResolveSacrificeUnlessDiscard(0, "missing-card")
		if err == nil {
			t.Fatal("expected nonexistent discard selection to fail")
		}
		if gs.PendingSacrificeUnlessDiscard == nil {
			t.Fatal("invalid discard selection consumed the pending choice")
		}
		if err := gs.ResolveSacrificeUnlessDiscard(0, ""); err != nil {
			t.Fatalf("decline discard: %v", err)
		}
		if _, err := gs.Battlefield.GetCard(src.ID); err == nil {
			t.Fatal("source survived after declining discard")
		}
	})

	t.Run("discard keeps source", func(t *testing.T) {
		gs := game.NewGame("P0", "P1", false)
		src := newCard(t)
		src.Owner, src.Controller = 0, 0
		gs.Battlefield.AddCard(src)

		discard := game.NewCreature("Factory NG Filler", game.ManaCost{}, 1, 1, 0)
		gs.Players[0].Hand.AddCard(discard)

		stackObj := &game.StackObject{
			IsAbility:   true,
			Text:        "sacrifice_unless_discard",
			Controller:  0,
			Card:        src,
		}
		if err := gs.ExecuteAbilityEffect(stackObj); err != nil {
			t.Fatalf("ExecuteAbilityEffect: %v", err)
		}
		if err := gs.ResolveSacrificeUnlessDiscard(0, discard.ID); err != nil {
			t.Fatalf("discard choice: %v", err)
		}
		if _, err := gs.Battlefield.GetCard(src.ID); err != nil {
			t.Fatalf("source was sacrificed despite discard: %v", err)
		}
		if gs.Players[0].Hand.HasCard(discard.ID) {
			t.Fatal("discarded card remains in hand")
		}
	})

	t.Run("unconditional sacrifice remains unconditional", func(t *testing.T) {
		gs := game.NewGame("P0", "P1", false)
		src := game.NewCreature("Factory NG Filter Source", game.ManaCost{}, 2, 2, 0)
		src.Owner, src.Controller = 0, 0
		gs.Battlefield.AddCard(src)

		stackObj := &game.StackObject{
			IsAbility:   true,
			Text:        "sacrifice",
			Controller:  0,
			Card:        src,
			EffectValue: map[string]interface{}{},
		}
		if err := gs.ExecuteAbilityEffect(stackObj); err != nil {
			t.Fatalf("ExecuteAbilityEffect: %v", err)
		}
		if gs.PendingSacrificeUnlessDiscard != nil {
			t.Fatal("unconditional sacrifice opened a discard choice")
		}
		if _, err := gs.Battlefield.GetCard(src.ID); err == nil {
			t.Fatal("unconditional sacrifice did not execute")
		}
	})
}
