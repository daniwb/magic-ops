package cards

import (
	"testing"

	"magic-backend/game"
)

func TestFactoryNGGainControlTargetAttached_5ac3995d10(t *testing.T) {
	def := &CardDefinition{
		Name: "Frenzied Fugue",
		Type: "Enchantment",
		Abilities: []AbilityDSL{{
			Type:    "triggered",
			Trigger: "etb",
			Effect:  "gain_control",
			Affects: &TargetFilter{Filter: "enchanted"},
			Value: map[string]interface{}{
				"until_eot": true,
				"untap":     true,
				"keywords":  []interface{}{"Haste"},
			},
		}},
		Status: StatusManual,
	}
	aura, err := def.ToGameCard(0)
	if err != nil {
		t.Fatalf("ToGameCard: %v", err)
	}

	gs := game.NewGame("P0", "P1", false)
	creature := game.NewCreature("Enchanted Permanent", game.ManaCost{}, 2, 2, 1)
	creature.Owner, creature.Controller = 1, 1
	creature.Tap()
	gs.Battlefield.AddCard(creature)

	aura.AttachedTo = creature.ID
	gs.Battlefield.AddCard(aura)
	gs.RegisterCardAbilities(aura)

	fire(t, gs, game.CreateETBEvent(aura, 0))

	if creature.Controller != 0 {
		t.Fatalf("attached permanent controller = %d, want 0", creature.Controller)
	}
	if creature.Tapped {
		t.Fatal("attached permanent remained tapped")
	}
	if !creature.HasKeywordWithEffects(game.Haste, gs.EffectManager) {
		t.Fatal("attached permanent did not gain haste")
	}

	gs.EventBus.Publish(game.NewEvent(game.EventCleanup))
	if creature.Controller != 1 {
		t.Fatalf("until-eot control was not reverted, controller = %d", creature.Controller)
	}
}
