package cards

// task-2427 (event_unsupported:deals_damage, receiver side) shape tests.
// The passive voice ("Whenever ~ IS DEALT damage" — the Enrage family, Apex
// Altisaur/Stuffy Doll class) names the damage RECEIVER, but every
// TriggerCondition subject filter matched event.Source (the DEALER), so the
// whole family was gated dead (reparse.py DEAD_TRIGGER_EVENTS). This round:
// slotparse splits "is dealt damage" into its own v2 event is_dealt_damage,
// the converter sets TriggerCondition.ReceiverSide, and Matches evaluates
// every subject filter against event.Target when the flag is set. Combat
// damage to a creature also dispatches through ProcessTriggeredAbilities now
// (combat.go previously discarded the matched record triggers).
// New file per the frozen-registry/shape-test convention.

import (
	"testing"

	"magic-backend/game"
)

func dealtDamageDef(name string, subject *TargetFilter) *CardDefinition {
	return &CardDefinition{
		Name: name,
		Type: "Creature", Types: []string{"Creature"},
		Power: intPtr(3), Toughness: intPtr(3),
		Abilities: []AbilityDSL{
			{Type: "triggered",
				TriggerV2: &TriggerSpec{Event: "is_dealt_damage", Subject: subject},
				Effects:   []EffectAtom{{Effect: "gain_life", Value: map[string]interface{}{"amount": float64(5)}}}},
		},
		Status: StatusManual,
	}
}

// "Whenever this creature is dealt damage, you gain 5 life." — Enrage/self family.
// Self DEALING damage must NOT fire (that's the dealer-side reading); self
// RECEIVING damage must.
func TestShape_Triggered_IsDealtDamageSelf(t *testing.T) {
	def := dealtDamageDef("Shape Enraged Saur", &TargetFilter{Filter: "self"})
	src, err := def.ToGameCard(0)
	if err != nil {
		t.Fatalf("ToGameCard: %v", err)
	}
	gs := game.NewGame("P0", "P1", false)
	gs.Battlefield.AddCard(src)
	gs.RegisterCardAbilities(src)

	other := game.NewCreature("Shape Bystander Beast", game.ManaCost{}, 2, 2, 1)
	other.Owner, other.Controller = 1, 1
	gs.Battlefield.AddCard(other)

	life0 := gs.Players[0].Life

	// Self is the DEALER (dealer-side reading) — must NOT fire.
	gs.ProcessTriggeredAbilities(game.CreateDamageEvent(src, other, 2, 0))
	drainStack(t, gs)
	if got := gs.Players[0].Life; got != life0 {
		t.Fatalf("receiver-side trigger fired on self DEALING damage (life %d -> %d)", life0, got)
	}
