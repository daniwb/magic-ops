// exileFromBattlefield moves a permanent from the battlefield directly to
// its owner's exile zone, mirroring exileFromHand/exileFromGraveyard's
// RemoveCard/AddCard/EventExiled shape (activated_abilities.go) for the
// battlefield zone specifically.
func (gs *GameState) exileFromBattlefield(card *Card) {
	if card == nil {
		return
	}
	gs.Battlefield.RemoveCard(card.ID) //nolint:errcheck
	owner := card.Owner
	if owner < 0 || owner >= len(gs.Players) {
		owner = card.Controller
	}
	if owner < 0 || owner >= len(gs.Players) {
		return
	}
	if gs.Players[owner].Exile != nil {
		gs.Players[owner].Exile.AddCard(card)
	}
	if gs.EventBus != nil {
		gs.EventBus.Publish(NewEvent(EventExiled).WithSource(card).WithPlayer(owner))
	}
}

// GetUntappedLands returns all untapped lands controlled by a specific player
func (z *Zone) GetUntappedLands(controller int) []*Card {
	if z.Type != ZoneBattlefield {
		return []*Card{}
	}
	lands := []*Card{}
	for _, card := range z.Cards {
		if card.IsLand() && card.Controller == controller && !card.Tapped {
			lands = append(lands, card)
		}
	}
	return lands
}

// UntapAll untaps all permanents controlled by a specific player (Rule 502 -
// Untap Step). Publishes EventUntapped for each permanent actually untapped
// (i.e. that was tapped beforehand) when the zone is wired to a GameState via
// SetGame ([VOCAB] #1683 — "whenever this creature becomes untapped" triggers,
// e.g. Ghostly Pilferer, previously only fired off explicit untap effects like
// Twiddle, never off the normal untap step).
func (z *Zone) UntapAll(controller int) {
	if z.Type != ZoneBattlefield {
		return
	}
	for _, card := range z.Cards {
		if card.Controller == controller {
			// Exert (Rule 701.39): skip exactly one untap step, then clear.
			if card.SkipNextUntap {
				card.SkipNextUntap = false
				continue
			}
			wasTapped := card.Tapped
			card.Untap()
			if wasTapped && z.game != nil && z.game.EventBus != nil {
				z.game.EventBus.Publish(NewEvent(EventUntapped).WithSource(card).WithPlayer(card.Controller))
			}
		}
	}
}

	target.Owner, target.Controller = 1, 1
	target.Tapped = true
	gs.Battlefield.AddCard(target)

	sibling.Owner, sibling.Controller = 1, 1
	sibling.Tapped = true
	gs.Battlefield.AddCard(sibling)

	}
