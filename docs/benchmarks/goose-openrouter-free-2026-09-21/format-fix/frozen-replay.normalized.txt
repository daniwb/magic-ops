package game

import (
	"testing"

	"magic-backend/cards"
)

// TestFactoryNGCreateTokenWithCanBlockOnlyFlying_59498a421f verifies that
// the generic create_token sequence honors abilities:"can_block_only_flying"
// by applying the token's permanent block restriction (Faerie Slumber Party
// class: "create two 1/1 blue Faerie creature tokens with flying and
// 'This token can block only creatures with flying.'").
func TestFactoryNGCreateTokenWithCanBlockOnlyFlying_59498a421f(t *testing.T) {
	// Register the new primitive under test
	_ = cards.V2EffectRegistry

	gs := NewGame("P0", "P1", false)

	// Create a source that triggers create_token with can_block_only_flying
	srcDef := &CardDefinition{
		Name: "Shape Faerie Swarm",
		Type: "Creature", Types: []string{"Creature"},
		Power: intPtr(1), Toughness: intPtr(1),
		Abilities: []AbilityDSL{
			{Type: "triggered",
				TriggerV2: &TriggerSpec{Event: "enters_battlefield"},
				Effects: []EffectAtom{{Effect: "create_token",
					Value: map[string]interface{}{
						"count":              1,
						"name":               "Faerie",
						"power":              1,
						"toughness":          1,
						"color":              "blue",
						"subtype":            "Faerie",
						"keywords":           []interface{}{"flying"},
						"can_block_only_flying": true,
					}}}},
		},
		Status: StatusManual,
	}
	card, err := srcDef.ToGameCard(0)
	if err != nil {
		t.Fatalf("ToGameCard: %v", err)
	}
	gs.Battlefield.AddCard(card)
	gs.RegisterCardAbilities(card)

	// Verify the source card is on battlefield
	if len(gs.Battlefield.GetAll()) == 0 {
		t.Fatal("source card not on battlefield")
	}

	// Resolve the triggered ability to create the token
	gs.ProcessTriggeredAbilities(CreateETBEvent(card, 0))
	drainStack(t, gs)

	// Find the created Faerie token
	var faerie *Card
	for _, c := range gs.Battlefield.GetAll() {
		if c.IsToken && c.Name == "Faerie" && c.Controller == 0 {
			faerie = c
			break
		}
	}
	if faerie == nil {
		t.Fatal("Faerie token was not created")
	}

	// Verify CanBlockOnlyFlying is set on the token
	if !faerie.CanBlockOnlyFlying {
		t.Fatal("Faerie token should have CanBlockOnlyFlying=true")
	}

	// Create a non-flying creature for the blocker test
	nonFlying := NewCreature("Shape Earth Spider", ManaCost{}, 1, 1, 0)
	nonFlying.Owner, nonFlying.Controller = 0, 0
	gs.Battlefield.AddCard(nonFlying)

	// Create a flying creature for the valid blocker test
	flying := NewCreature("Shape Cloud Ray", ManaCost{}, 1, 1, 0)
	flying.Owner, flying.Controller = 0, 0
	flying.Keywords = append(flying.Keywords, Flying)
	gs.Battlefield.AddCard(flying)

	// Test 1: Faerie token (CanBlockOnlyFlying) cannot block non-flying creature
	if gs.canBlockOnlyFlyingBlockCheck(faerie, nonFlying) {
		t.Fatal("Faerie token with CanBlockOnlyFlying should NOT be able to block non-flying creature")
	}

	// Test 2: Faerie token (CanBlockOnlyFlying) CAN block flying creature
	if !gs.canBlockOnlyFlyingBlockCheck(faerie, flying) {
		t.Fatal("Faerie token with CanBlockOnlyFlying SHOULD be able to block flying creature")
	}

	// Test 3: A normal token without CanBlockOnlyFlying CAN block non-flying
	normalToken := NewToken("Shape Wolf", 2, 2, 0)
	gs.Battlefield.AddCard(normalToken)
	if !gs.canBlockOnlyFlyingBlockCheck(normalToken, nonFlying) {
		t.Fatal("Normal token without CanBlockOnlyFlying should be able to block non-flying creature")
	}
}

func drainStack(t *testing.T, gs *GameState) {
	for gs.Stack.Len() > 0 {
		gs.Step()
	}
}
