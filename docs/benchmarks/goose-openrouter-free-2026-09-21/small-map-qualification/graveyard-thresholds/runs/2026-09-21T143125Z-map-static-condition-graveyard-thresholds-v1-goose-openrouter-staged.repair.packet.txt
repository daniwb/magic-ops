def parse_static_condition(text):
    """single 'as long as' clause -> Condition dict, or None if it isn't one
    of the safely-executable shapes."""
    text = text.strip().rstrip(',').strip()
    if not text:
        return None

    # ticket #5261: Angelic Voices — the runtime's existing control condition
    # supports a less-than threshold plus negative type/color filters.
    if text == 'you control no nonartifact, nonwhite creatures':
        return {'kind': 'control',
                'filter': {'filter': 'creature', 'exclude_type': 'artifact',
                           'exclude_color': 'w'},
                'count': '1_or_more', 'cmp': 'lt'}

    # ticket #5320 (static_conditional / "as long as" family): CTRL_OR_RE
    # must be tried BEFORE CTRL_RE — CTRL_RE's own qualifier group greedily
    # matches the leading "a"/"an" of "you control a Goblin or Orc" and then
    # explicitly returns None once its rest-parsing (parse_control_filter,
    # which rejects any tail containing the word "or") fails, short-
    # circuiting before this branch would otherwise run.
    m = CTRL_OR_RE.match(text)
    if m:
        qual, w1, w2 = m.groups()
        other = qual is not None and qual.strip() == 'another'
        filt1 = parse_control_filter(w1)
        filt2 = parse_control_filter(w2)
        if filt1 is not None and filt2 is not None:
            if other:
                filt1 = dict(filt1, other=True)
                filt2 = dict(filt2, other=True)
            return {'any_of': [
                {'kind': 'control', 'filter': filt1, 'count': 'a'},
                {'kind': 'control', 'filter': filt2, 'count': 'a'},
            ]}

    m = CTRL_RE.match(text)
    if m:
        qual, at_least_n, or_more_n, rest = m.groups()
        count = _cond_count(qual) if qual in ('a', 'an', 'no') else None
        if count is None:
            n = at_least_n or or_more_n
            count = _cond_count(n) if n else None
        if count is None:
            return None
        filt = parse_control_filter(rest)
        if filt is not None:
            return {'kind': 'control', 'filter': filt, 'count': count}
        # task-3090 (static_conditional): "you control a <color> <type>"
        # (Blazethorn Scarecrow "you control a red creature"/"a green
        # creature") — parse_control_filter only resolves bare subtypes, not
        # color words, so route color-qualified filters onto control_color
        # (colored_control_filter primitive, ticket #3088; game/condition.go
        # already resolves Color against the SAME Filter/count fields the
        # plain "control" kind uses here, just scoped to one color).
        cm = re.match(r'^(white|blue|black|red|green) ([a-z]+)$', rest.strip())
        if cm:
            typ = cm.group(2).rstrip('s')
            if typ in _CTRL_TYPE_WORDS:
                return {'kind': 'control_color', 'filter': {'filter': typ},
                        'count': count, 'color': COLOR_LETTER[cm.group(1)]}
        return None

    # ticket #3820 (static_conditional): "you control another <type>" (Court
    # Homunculus "you control another artifact") — CTRL_RE's qualifier group
    # only recognizes "a"/"an"/"no"/"at least N"/"N or more", never the bare
    # "another" quantifier (mirrors CTRL_ANOTHER_COLOR_RE's identical gap for
    # the color-qualified sibling just below), so this never reaches CTRL_RE.
    # Routes onto the SAME plain 'control' kind CTRL_RE already emits, count
    # 'a' plus filter.other=True (task 2367) so the source's own instance
    # doesn't trivially satisfy its own condition.
    m = re.match(r'^you control another ([a-z]+)$', text)
    if m:
        word = m.group(1).rstrip('s')
        if word in _CTRL_TYPE_WORDS:
            return {'kind': 'control', 'filter': {'filter': word, 'other': True}, 'count': 'a'}

    m = DEFENDING_PLAYER_CTRL_RE.match(text)
    if m:
        qual, at_least_n, or_more_n, rest = m.groups()
        count = _cond_count(qual) if qual in ('a', 'an', 'no') else None
        if count is None:
            n = at_least_n or or_more_n
            count = _cond_count(n) if n else None
        if count is None:
            return None
        filt = parse_control_filter(rest)
        if filt is not None:
            return {'kind': 'control', 'who': 'defending_player',
                    'filter': filt, 'count': count}
        cm = re.match(r'^(white|blue|black|red|green) ([a-z]+)$', rest.strip())
        if cm:
            typ = cm.group(2).rstrip('s')
            if typ in _CTRL_TYPE_WORDS:
                return {'kind': 'control_color', 'who': 'defending_player',
                        'filter': {'filter': typ}, 'count': count,
                        'color': COLOR_LETTER[cm.group(1)]}
        return None

    m = OPPONENT_CTRL_COLOR_RE.match(text)
    if m:
        who, color, typ = m.groups()
        typ = typ.rstrip('s')
        if typ in _CTRL_TYPE_WORDS:
            cond = {'kind': 'control_color', 'filter': {'filter': typ},
                    'count': 'a', 'color': COLOR_LETTER[color]}
            if who == 'an opponent':
                cond['who'] = 'opponent'
            return cond

    # ticket #4968 (static_conditional): "an opponent is poisoned" means that
    # player has one or more poison counters. Reuse the existing player-counter
    # threshold condition vocabulary; only the Oracle phrasing was missing.
    if text == 'an opponent is poisoned':
        return {'kind': 'player_counters', 'who': 'opponent', 'what': 'poison',
                'amount': 1}

    m = STATE_SELF_RE.match(text)
    if m:
        state = 'ring_bearer' if m.group(2) == 'your ring-bearer' else m.group(2)
        return {'kind': 'state', 'of': 'self', 'state': state}

    m = STATE_ATTACHED_COUNT_RE.match(text)
    if m:
        amt, kind_word = m.groups()
        n = _amt(amt)
        if n is not None:
            state = 'equipped' if kind_word == 'equipment' else 'enchanted'
            return {'kind': 'state', 'of': 'self', 'state': state, 'amount': n}

    m = STATE_SELF_COUNTERS_RE.match(text)
    if m:
        qual, at_least_n, or_more_n, fewer_n, ctype = m.groups()
        if fewer_n is not None:
            n = _amt(fewer_n)
            if n is None:
                return None
            return {'kind': 'state', 'of': 'self', 'state': 'has_counters', 'what': ctype,
                    'cmp': 'lt', 'amount': n}
        if qual in ('a', 'an'):
            n = 1
        elif qual == 'no':
            return {'kind': 'state', 'of': 'self', 'state': 'has_counters', 'what': ctype,
                    'cmp': 'lt', 'amount': 1}
        else:
            n = _amt(at_least_n or or_more_n)
        if n is None:
            return None
        return {'kind': 'state', 'of': 'self', 'state': 'has_counters', 'what': ctype, 'amount': n}

    m = GY_COUNT_RE.match(text)
    if m:
        who, amt = m.groups()
        n = _amt(amt)
        if n is None:
            return None
        cond = {'kind': 'threshold', 'zone': 'graveyard', 'amount': n}
        if who == 'an opponent':
            cond['who'] = 'opponent'
        return cond

    m = GY_COUNT_THERE_RE.match(text)
    if m:
        n = _amt(m.group(1))
        if n is None:
            return None
        return {'kind': 'threshold', 'zone': 'graveyard', 'amount': n}

    m = GY_COUNT_THERE_FILTERED_RE.match(text)
    if m:
        amt, word = m.groups()
        n = _amt(amt)
        what = _put_bf_card_type(word)
        if n is not None and what is not None and what != 'card':
            return {'kind': 'threshold', 'zone': 'graveyard', 'what': what, 'amount': n}

    m = GY_COUNT_THERE_SINGULAR_RE.match(text)
    if m:
        what = _put_bf_card_type(m.group(1))
        if what is not None and what != 'card':
            return {'kind': 'threshold', 'zone': 'graveyard', 'what': what, 'amount': 1}

    # ticket #5318 (static_conditional / "gets as long as" family): "an
    # opponent owns a card in exile" (Warden of the Beyond) — the exile-zone
    # sibling of the graveyard threshold kind; game/condition.go's
    # "threshold" case already special-cases c.Zone == "exile" to count
    # who.Exile instead of who.Graveyard, so only the parse side was missing.
    if text == 'an opponent owns a card in exile':
        return {'kind': 'threshold', 'zone': 'exile', 'who': 'opponent', 'amount': 1}

    m = CTRL_COLOR_OR_RE.match(text)
    if m:
        c1, c2 = m.groups()
        return {'any_of': [
            {'kind': 'control_color', 'filter': {'filter': 'permanent'}, 'color': COLOR_LETTER[c1]},
            {'kind': 'control_color', 'filter': {'filter': 'permanent'}, 'color': COLOR_LETTER[c2]},
        ]}

    m = CTRL_ANOTHER_COLOR_RE.match(text)
    if m:
        color, typ = m.groups()
        typ = typ.rstrip('s')
        if typ in _CTRL_TYPE_WORDS:
            return {'kind': 'control_color', 'filter': {'filter': typ, 'other': True},
                    'count': 'a', 'color': COLOR_LETTER[color]}

    m = CTRL_ANOTHER_RE.match(text)
    if m:
        filt = parse_control_filter('another ' + m.group(1))
        if filt is not None:
            return {'kind': 'control', 'filter': filt, 'count': 'a'}

    m = OPPONENT_CTRL_NONE_RE.match(text)
    if m:
        filt = parse_control_filter(m.group(1))
        if filt is not None:
            return {'kind': 'control', 'who': 'opponent', 'filter': filt, 'count': 'no'}

    m = OPPONENT_CTRL_A_RE.match(text)
    if m:
        filt = parse_control_filter(m.group(1))
        if filt is not None:
            return {'kind': 'control', 'who': 'opponent', 'filter': filt, 'count': 'a'}

    m = ANY_PLAYER_CTRL_COLOR_RE.match(text)
    if m:
        color, typ = m.groups()
        typ = typ.rstrip('s')
        if typ in _CTRL_TYPE_WORDS:
            return {'kind': 'control_color', 'who': 'any_player',
                    'filter': {'filter': typ}, 'count': 'a', 'color': COLOR_LETTER[color]}

    m = _GY_TWO_TYPE_RE.match(text)
    if m:
        t1, t2 = m.group(1), m.group(2)
        if t1 in _GY_BARE_TYPE_WORDS and t2 in _GY_BARE_TYPE_WORDS:

                {'kind': 'threshold', 'zone': 'graveyard', 'what': t1, 'amount': 1},
                {'kind': 'threshold', 'zone': 'graveyard', 'what': t2, 'amount': 1},

    m = HAND_RE.match(text)
    if m:
        amt, direction = m.groups()
        n = _amt(amt)
        if n is None:
            return None
        return {'kind': 'hand', 'cmp': 'lte' if direction == 'fewer' else 'gte', 'amount': n}

    if HAND_NONE_RE.match(text):
        return {'kind': 'hand', 'cmp': 'lte', 'amount': 0}

    m = CAST_TURN_RE.match(text)
    if m:
        qual, at_least_n, or_more_n, kind_word = m.groups()
        n = 1 if qual == 'a' or qual == 'an' else _amt(at_least_n or or_more_n)
        if n is None:
            return None
        what = 'you_cast_spell'

            what = 'you_cast_noncreature_spell'

            what = 'you_cast_creature_spell'

    m = STATE_ATTACH_RE.match(text)
    if m:
        of = 'equipped' if m.group(1) == 'equipped creature' else 'enchanted'
        word = m.group(2)
        state = 'color_' + COLOR_LETTER[word] if word in COLOR_LETTER else word
        return {'kind': 'state', 'of': of, 'state': state}

    m = LIFE_RE.match(text)
    if m:
        who, amt, direction = m.groups()
        n = _amt(amt)
        if n is None:
            return None
        cond = {'kind': 'life', 'cmp': 'lte' if direction in ('less', 'fewer') else 'gte', 'amount': n}
        if who == 'an opponent':
            cond['who'] = 'opponent'
        return cond

    m = LIFE_HALF_STARTING_RE.match(text)
    if m:
        who, lte = m.groups()
        cond = {'kind': 'life', 'cmp': 'lte' if lte else 'lt',
                'baseline': 'starting', 'fraction': 2}
        if who != 'your':
            cond['who'] = 'opponent'
        return cond

    m = LIFE_GT_STARTING_RE.match(text)
    if m:
        who = m.group(1)
        cond = {'kind': 'life', 'cmp': 'gt', 'baseline': 'starting'}
        if who != 'your':
            cond['who'] = 'opponent'
        return cond

    m = LIFE_N_GT_STARTING_RE.match(text)
    if m:
        who, amt = m.groups()
        n = _amt(amt)
        if n is None:
            return None
        cond = {'kind': 'life', 'cmp': 'gte', 'baseline': 'starting', 'amount': n}
        if who != 'your':
            cond['who'] = 'opponent'
        return cond

    m = LIFE_N_MORE_STARTING_RE.match(text)
    if m:
        n = _amt(m.group(1))
        if n is None:
            return None
        return {'kind': 'life', 'cmp': 'gte', 'baseline': 'starting', 'amount': n}

    m = TURN_RE.match(text)
    if m:
        return {'kind': 'turn', 'whose': 'yours' if m.group(1) == 'your turn' else 'opponents'}

    # ticket:map.static-condition-gained-life-pump/v1
    if text == 'you gained life this turn':
        return {'kind': 'happened', 'what': 'you_gained_life'}

    # ticket #5320 (static_conditional / "as long as" family): "you've lost
    # life this turn" (Essence Channeler) — the you_lost_life sibling of
    # you_gained_life just above; game/condition.go's "happened" case already
    # evaluates it off player.LifeLostThisTurn (added for the negated
    # "if you didn't lose life this turn" idiom), only this positive
    # as-long-as phrasing was never parsed.
    if text == "you've lost life this turn":
        return {'kind': 'happened', 'what': 'you_lost_life'}

    # ticket #5320 (static_conditional / "as long as" family): "an opponent
    # controls more lands than you" (Verge Rangers) — the existing 'compare'
    # kind (game/condition.go: What "lands"/"creatures", Who "opponent"
    # flips mine/theirs) already evaluates this; only the parse side was
    # missing.
    if text == 'an opponent controls more lands than you':
        return {'kind': 'compare', 'what': 'lands', 'who': 'opponent'}

    # ticket:map.static-condition-drawn-two-keywords/v3
    if text == 'you\'ve drawn two or more cards this turn':
        return {'kind': 'happened', 'what': 'you_drew_cards', 'amount': 2}

    # ticket #5048 (static_conditional): "N or more nonland permanents
    # entered the battlefield under your control this turn" (Armory Mice,
    # Celebration cycle) — the happened-kind what the Go runtime already
    # evaluates (game/condition.go nonland_permanent_entered_battlefield,
    # ticket #3479) but parse_static_condition never recognized, so the
    # whole "gets +0/+2 as long as ..." sentence stayed an honest miss even
    # though both halves were individually executable.
    m = re.match(
        r"^(\w+) or more nonland permanents entered the battlefield "
        r"under your control this turn$", text)
    if m:
        n = _amt(m.group(1))
        if n is not None:
            return {'kind': 'happened',
                    'what': 'nonland_permanent_entered_battlefield',
                    'amount': n}
    if DELIRIUM_RE.match(text):
        return {'kind': 'named', 'name': 'delirium'}
    if re.match(r"^you'?re the monarch$", text):
        return {'kind': 'named', 'name': 'monarch'}
    if re.match(r"^you have the city'?s blessing$", text):
        return {'kind': 'named', 'name': 'citys_blessing'}

    # ticket:map.static-condition-self-attacked-keyword/v1 (extended,
    # ticket #5320: The Lunar Whale phrases the same self-referential clause
    # via its own printed name rather than "it" — apply_self_ref already
    # normalizes that to '~' upstream, so the tilde form reaches here too).
    if text in ('it attacked this turn', '~ attacked this turn'):
        return {'kind': 'happened', 'what': 'self_attacked_this_turn'}

    return None
